RATE = 50
BURST = 200


def allow(key, now):
    bucket = buckets.get(key)
    refill(bucket, now, RATE)
    return bucket.take(1)


def refill(bucket, now, rate):
    bucket.tokens = min(BURST, bucket.tokens + (now - bucket.ts) * rate)
    bucket.ts = now
