$ git config set filter.nbstrip.clean nbstrip-not-installed
$ git add notebooks/01-error-analysis.ipynb

error: cannot fork to run external filter 'nbstrip-not-installed'

$ git status --short

error: cannot fork to run external filter 'nbstrip-not-installed'
